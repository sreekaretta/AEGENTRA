FILE 4: agents/base_agent.py
PURPOSE: Abstract base class (Template Method pattern) used by all 7 pipeline
agents. Sets up a KafkaConsumer (manual commit, group_id = agent_name + "-group")
and optional KafkaProducer. The run() loop calls the abstract process_message(),
forwards non-None results to the output topic, then commits offsets.
================================================================================

"""
Base Agent for Aegentra Multi-Agent System
"""

from kafka import KafkaConsumer, KafkaProducer
import json
import logging

logging.basicConfig(level=logging.INFO)
logger = logging.getLogger(__name__)


class BaseAgent:
    """Base class for all agents in the Aegentra system"""

    def __init__(self, agent_name, input_topic, output_topic=None):
        self.agent_name = agent_name
        self.input_topic = input_topic
        self.output_topic = output_topic

        self.consumer = KafkaConsumer(
            input_topic,
            bootstrap_servers=['localhost:9092'],
            value_deserializer=lambda m: json.loads(m.decode('utf-8')),
            group_id=f"{self.agent_name}-group",
            auto_offset_reset='earliest',
            enable_auto_commit=False,
            max_poll_interval_ms=300000,
            session_timeout_ms=30000
        )

        self.producer = None
        if output_topic:
            self.producer = KafkaProducer(
                bootstrap_servers=['localhost:9092'],
                value_serializer=lambda x: json.dumps(x).encode('utf-8'),
                request_timeout_ms=30000
            )

        logger.info(f"{agent_name} initialized")
        logger.info(f"Consuming from: {input_topic}")
        if output_topic:
            logger.info(f"Producing to: {output_topic}")

    def deserialize_message(self, message):
        return message.value

    def process_message(self, message):
        raise NotImplementedError("Subclasses must implement process_message()")

    def run(self):
        logger.info(f"{self.agent_name} starting...")

        try:
            for message in self.consumer:
                decoded_message = self.deserialize_message(message)

                if decoded_message:
                    msg_id = decoded_message.get('incident_id', decoded_message.get('alert_id', 'unknown'))
                    logger.info(f"Received message: {msg_id}")

                    result = self.process_message(decoded_message)

                    if result and self.producer:
                        self.producer.send(self.output_topic, result)
                        self.producer.flush(timeout=10)

                    self.consumer.commit()

        except KeyboardInterrupt:
            logger.info(f"{self.agent_name} stopping...")
            self.cleanup()
        except Exception as e:
            logger.error(f"Error in {self.agent_name}: {e}")
            import traceback
            traceback.print_exc()
        finally:
            self.cleanup()

    def cleanup(self):
        logger.info("Closing Kafka connections...")
        if self.consumer:
            self.consumer.close()
        if self.producer:
            self.producer.close()

FILE 5: agents/monitoring/monitoring_agent.py
PURPOSE: Pipeline Stage 1. Consumes "web-detection-alerts", applies a 3-rule
filter: HIGH/MEDIUM (severity>=2) → forward; LOW in 12-item safe list → forward
as safe_config_fix; everything else → "filtered-out-alerts". Tracks stats and
logs automation rate every 50 messages.
================================================================================

import sys
import os
sys.path.append(os.path.dirname(os.path.dirname(os.path.abspath(__file__))))

from base_agent import BaseAgent
import logging
from datetime import datetime

logging.basicConfig(level=logging.INFO)
logger = logging.getLogger(__name__)


class WebMonitoringAgent(BaseAgent):
    """Smart Web Monitoring Agent with Filtered Alerts Topic"""

    def __init__(self):
        super().__init__(
            agent_name="WebMonitor-Enterprise-v8",
            input_topic="web-detection-alerts",
            output_topic="unified-alerts"
        )

        self.safe_low_severity = [
            'X-Content-Type-Options Header Missing',
            'X-Frame-Options Header Not Set',
            'X-XSS-Protection Header Not Set',
            'Strict-Transport-Security Header Not Set',
            'Content-Security-Policy Header Missing',
            'Directory Browsing',
            'Server Leaks Version Information',
            'Cookie Without Secure Flag',
            'Cookie Without HttpOnly Flag',
            'Missing Anti-clickjacking Header',
            'Referrer Policy Header Not Set',
            'Permissions Policy Header Not Set'
        ]

        self.stats = {'total': 0, 'high_medium': 0, 'safe_low': 0, 'filtered': 0}

    def process_message(self, message):
        self.stats['total'] += 1

        try:
            severity = message.get('severity', 0)
            vulnerability = message.get('vulnerability', '')

            if severity >= 2:
                self.stats['high_medium'] += 1
                message['detection_source'] = 'web'
                message['automation_reason'] = 'high_medium_severity'
                logger.info(f"✅ HIGH/MED: {vulnerability[:50]} (severity: {severity})")
                return message

            elif self.is_safe_low_severity(vulnerability):
                self.stats['safe_low'] += 1
                message['detection_source'] = 'web'
                message['automation_reason'] = 'safe_config_fix'
                logger.info(f"✅ SAFE-LOW: {vulnerability[:50]} (severity: {severity})")
                return message

            else:
                self.stats['filtered'] += 1
                filtered_alert = {
                    'original_alert': message,
                    'filtered_at': datetime.now().isoformat(),
                    'filter_reason': 'Low severity + not safe for automation',
                    'severity': severity,
                    'vulnerability': vulnerability,
                    'url': message.get('url', ''),
                    'automation_decision': 'FILTERED',
                    'requires_manual_review': False
                }
                if self.producer:
                    self.producer.send('filtered-out-alerts', filtered_alert)
                logger.info(f"❌ FILTERED: {vulnerability[:50]} → filtered-out-alerts")
                return None

        except Exception as e:
            logger.error(f"Error processing message: {e}")
            return None

    def is_safe_low_severity(self, vulnerability):
        for safe_vuln in self.safe_low_severity:
            if safe_vuln.lower() in vulnerability.lower():
                return True
        return False

    def run(self):
        logger.info(f"\n{'='*60}")
        logger.info(f"  {self.agent_name} STARTED - ENTERPRISE FILTERING")
        logger.info(f"{'='*60}")
        logger.info("Filtering Rules:")
        logger.info("  ✅ High/Medium severity → unified-alerts")
        logger.info("  ✅ Low severity + safe config → unified-alerts")
        logger.info("  ❌ Low severity + risky → filtered-out-alerts\n")

        try:
            for kafka_message in self.consumer:
                message = kafka_message.value
                output = self.process_message(message)

                if output and self.producer:
                    self.producer.send(self.output_topic, output)
                    self.producer.flush(timeout=10)

                self.consumer.commit()

                if self.stats['total'] % 50 == 0:
                    total_forwarded = self.stats['high_medium'] + self.stats['safe_low']
                    automation_rate = (total_forwarded / self.stats['total'] * 100) if self.stats['total'] > 0 else 0
                    logger.info(
                        f"\n📊 PROGRESS - Processed: {self.stats['total']}"
                        f"\n   Forwarded: {total_forwarded} ({automation_rate:.1f}%)"
                        f"\n   Filtered: {self.stats['filtered']}\n"
                    )

        except KeyboardInterrupt:
            self.print_final_stats()
        finally:
            self.cleanup()

    def print_final_stats(self):
        total_forwarded = self.stats['high_medium'] + self.stats['safe_low']
        automation_rate = (total_forwarded / self.stats['total'] * 100) if self.stats['total'] > 0 else 0
        logger.info(f"\n📊 FINAL STATISTICS:")
        logger.info(f"Total Processed:        {self.stats['total']}")
        logger.info(f"  High/Medium Severity: {self.stats['high_medium']}")
        logger.info(f"  Safe Low-Severity:    {self.stats['safe_low']}")
        logger.info(f"  Total Forwarded:      {total_forwarded} ({automation_rate:.1f}%)")
        logger.info(f"  Filtered Out:         {self.stats['filtered']}")


if __name__ == "__main__":
    agent = WebMonitoringAgent()
    agent.run()

FILE 6: agents/correlation/correlation_agent.py
PURPOSE: Pipeline Stage 2. Groups alerts into incidents using key = source_ip +
"_" + alert_type. Creates incident when group reaches threshold (default 5,
from config.yaml). Incident ID format: "INC-{timestamp}-{count}". Flushes old
groups exceeding the 5-minute time window. On shutdown, flushes all remaining
partial groups via cleanup().
================================================================================

import sys
import os
sys.path.append(os.path.dirname(os.path.dirname(os.path.abspath(__file__))))

from base_agent import BaseAgent
from datetime import datetime, timedelta
from collections import defaultdict
import logging
import yaml

logging.basicConfig(level=logging.INFO)
logger = logging.getLogger(__name__)


class CorrelationAgent(BaseAgent):
    """ENTERPRISE Correlation Agent: Groups related alerts into incidents"""

    def __init__(self):
        super().__init__(
            agent_name="CorrelationAgent-Enterprise-v8",
            input_topic="unified-alerts",
            output_topic="correlated-incidents"
        )

        config_path = os.path.join(os.path.dirname(__file__), '..', '..', 'config.yaml')
        with open(config_path, 'r') as f:
            config = yaml.safe_load(f)

        correlation_config = config.get('correlation', {})
        time_window_minutes = correlation_config.get('time_window_minutes', 5)
        self.group_threshold = correlation_config.get('group_threshold', 5)
        self.max_alerts = correlation_config.get('max_alerts_per_incident', 100)

        self.time_window = timedelta(minutes=time_window_minutes)
        self.alert_buffer = defaultdict(list)
        self.incident_count = 0

        logger.info(f"✅ Correlation settings: window={time_window_minutes}min, threshold={self.group_threshold}")

    def process_message(self, message):
        try:
            alert = message
            source_ip = alert.get('source_ip', alert.get('url', 'unknown'))
            alert_type = alert.get('alert_type', 'web_vulnerability')
            timestamp_str = alert.get('timestamp', datetime.now().isoformat())

            try:
                timestamp = datetime.fromisoformat(timestamp_str)
            except:
                timestamp = datetime.now()

            group_key = f"{source_ip}_{alert_type}"
            self.alert_buffer[group_key].append({'alert': alert, 'timestamp': timestamp})

            logger.info(f"Added alert to group {group_key}. Buffer size: {len(self.alert_buffer[group_key])}")

            if len(self.alert_buffer[group_key]) >= self.group_threshold:
                incident = self.create_incident(group_key)
                return incident

            self.flush_old_groups()
            return None

        except Exception as e:
            logger.error(f"Error processing message: {e}")
            import traceback
            traceback.print_exc()
            return None

    def create_incident(self, group_key):
        alerts = self.alert_buffer[group_key]

        if not alerts:
            return None

        if len(alerts) > self.max_alerts:
            alerts = alerts[:self.max_alerts]

        alerts.sort(key=lambda x: x['timestamp'])
        self.incident_count += 1

        incident = {
            'incident_id': f"INC-{int(datetime.now().timestamp())}-{self.incident_count}",
            'created_at': datetime.now().isoformat(),
            'group_key': group_key,
            'alert_count': len(alerts),
            'first_seen': alerts[0]['timestamp'].isoformat(),
            'last_seen': alerts[-1]['timestamp'].isoformat(),
            'source_ip': alerts[0]['alert'].get('source_ip', alerts[0]['alert'].get('url', 'unknown')),
            'alert_type': alerts[0]['alert'].get('alert_type', 'web_vulnerability'),
            'alerts': [a['alert'] for a in alerts],
            'severity': max([a['alert'].get('severity', 0) for a in alerts]),
            'summary': f"{len(alerts)} related alerts from {group_key}"
        }

        logger.info(f"✅ Created incident {incident['incident_id']} with {len(alerts)} alerts")

        if self.producer:
            self.producer.send(self.output_topic, incident)
            self.producer.flush(timeout=10)

        self.alert_buffer[group_key] = []
        return incident

    def flush_old_groups(self):
        now = datetime.now()
        for group_key in list(self.alert_buffer.keys()):
            alerts = self.alert_buffer[group_key]
            if not alerts:
                continue
            oldest = min([a['timestamp'] for a in alerts])
            if now - oldest > self.time_window:
                if len(alerts) > 0:
                    self.create_incident(group_key)

    def cleanup(self):
        logger.info("Flushing all remaining groups...")
        total_flushed = 0
        for group_key in list(self.alert_buffer.keys()):
            if len(self.alert_buffer[group_key]) > 0:
                incident = self.create_incident(group_key)
                if incident:
                    total_flushed += 1
        logger.info(f"Flushed {total_flushed} remaining incidents")
        logger.info(f"Total incidents created: {self.incident_count}")
        super().cleanup()

TASK:
Analyze this security incident and provide:
1. Threat Type, 2. Severity (1-10), 3. Confidence (0.0-1.0),
4. MITRE ATT&CK Technique ID, 5. MITRE Tactic, 6. Immediate Actions, 7. Reasoning

RESPOND ONLY WITH VALID JSON:
{{
  "threat_type": "...",
  "severity": 8,
  "confidence": 0.95,
  "mitre_technique": "T1190",
  "mitre_tactic": "Initial Access",
  "immediate_actions": ["add_waf_rule", "alert_security_team"],
  "reasoning": "..."
}}"""
        return prompt

    def parse_response(self, response_text, incident):

                'tech_stack': tech_stack,
                'actions': action_plans,
                'total_actions': len(action_plans),
                'reasoning': f"Universal mapping: {len(actions)} actions for {analysis.get('threat_type')} on {tech_stack['web_server']}",
                'universal_plan': True,
                'bias_free': True,
                'enterprise_grade': True
            }

            self.plan_count += 1
            logger.info(f"✅ Plan {self.plan_count}: {response_plan['plan_id']} with {len(action_plans)} actions")

            return response_plan

        except Exception as e:
            logger.error(f"Error creating response plan: {e}")
            import traceback
            traceback.print_exc()
            return None

    def _get_target_from_incident(self, incident: Dict) -> str:
        alerts = incident.get('alerts', [])
        if alerts:
            return alerts[0].get('url', 'unknown')
        return 'unknown'

    def _generate_parameters(self, action: str, analysis: Dict, tech_stack: Dict) -> Dict:
        params = {
            'threat_type': analysis.get('threat_type'),
            'severity': analysis.get('severity'),
            'confidence': analysis.get('confidence'),
            'cwe_id': analysis.get('cwe_id', '0')
        }
        if 'waf' in action:
            params['pattern'] = self._generate_waf_pattern(analysis.get('threat_type'))
            params['description'] = f"Block {analysis.get('threat_type')}"
        elif 'header' in action:
            params['headers'] = self._get_security_headers(analysis.get('threat_type'))
        elif 'rate' in action:
            params['requests_per_minute'] = 60 if analysis.get('severity', 0) >= 7 else 120
        elif 'alert' in action:
            params['priority'] = 'high' if analysis.get('severity', 0) >= 8 else 'medium'
        elif 'validation' in action or 'sanitize' in action:
            params['validation_type'] = self._get_validation_type(analysis.get('threat_type'))
        return params

    def _generate_waf_pattern(self, threat_type: str) -> str:
        patterns = {
            'SQL Injection': r"(\\'|\\\"|\\OR\\|\\UNION\\|\\SELECT\\|\\DROP\\)",
            'XSS': r"(<script|javascript:|onerror=|onload=|<img)",
            'Cross-Site Scripting': r"(<script|javascript:|onerror=|onload=|<img)",
            'Command Injection': r"(;|&&|\\||`|\\$\\()",
            'Path Traversal': r"(\\.\\.\\/|\\.\\.\\\\)",
            'Directory Traversal': r"(\\.\\.\\/|\\.\\.\\\\)",
            'XXE': r"(<!ENTITY|<!DOCTYPE|SYSTEM)",
            'SSRF': r"(file://|dict://|gopher://)"
        }
        return patterns.get(threat_type, r"(malicious|attack|exploit)")

    def _get_security_headers(self, threat_type: str) -> Dict:
        headers = {
            'X-Frame-Options': 'SAMEORIGIN',
            'X-Content-Type-Options': 'nosniff',
            'X-XSS-Protection': '1; mode=block'
        }
        if 'XSS' in threat_type or 'Cross-Site Scripting' in threat_type:
            headers['Content-Security-Policy'] = "default-src 'self'; script-src 'self'"
        if 'Clickjacking' in threat_type:
            headers['X-Frame-Options'] = 'DENY'
        return headers

    def _get_validation_type(self, threat_type: str) -> str:
        if 'SQL' in threat_type: return 'sql_injection'
        elif 'XSS' in threat_type or 'Cross-Site Scripting' in threat_type: return 'xss'
        elif 'Command' in threat_type: return 'command_injection'
        elif 'Path' in threat_type or 'Traversal' in threat_type: return 'path_traversal'
        else: return 'generic'

    def _calculate_priority(self, metadata: Dict, analysis: Dict) -> int:
        base_priority = 5
        if analysis.get('severity', 0) >= 8:
            base_priority += 3
        if analysis.get('confidence', 0) >= 0.9:
            base_priority += 1
        if metadata.get('risk_level') == 'high':
            base_priority -= 2
        return max(1, min(10, base_priority))

    def _estimate_duration(self, action: str, metadata: Dict) -> str:
        if metadata.get('category') == 'configuration': return '< 1 minute'
        elif metadata.get('category') == 'network_control': return '1-5 minutes'
        elif metadata.get('category') == 'code_fix': return '30-60 minutes'
        else: return '5-15 minutes'

    def cleanup(self):
        logger.info(f"\nTotal Plans Created: {self.plan_count}")
        super().cleanup()


if __name__ == "__main__":
    agent = UniversalResponsePlanningAgent()
    agent.run()
