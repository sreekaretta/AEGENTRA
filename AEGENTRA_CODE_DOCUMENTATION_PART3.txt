WHAT IT DOES:
  Full system restart script for AEGENTRA. Stops all 7 pipeline agents using pkill,
  restarts Kafka (docker stop/rm + docker run with --network host), restarts OWASP ZAP
  (same pattern), rebuilds the minimal-test-app via docker-compose, optionally starts
  Juice Shop on port 9090, offers to purge all Kafka topics (calls purge_all_topics.sh),
  then prints the exact terminal commands to launch all 7 agents + ZAP scanner.
  It does NOT auto-start agents — instead it prints commands for 8 separate terminals.

KEY BEHAVIOR:
  - 8-step sequence with interactive prompt at Step 6 (topic purge: y/n)
  - Kafka starts with --network host and KAFKA_ADVERTISED_LISTENERS=PLAINTEXT://localhost:9092
  - ZAP starts with api.disablekey=true so no API key is required
  - Minimal test app started via: cd ~/Aegentra1/minimal-test-app && docker-compose up -d
  - Final verification section shows all docker ps checks and Kafka topic list command

WHAT IT DOES:
  Wipes and recreates all 8 core AEGENTRA Kafka topics for a clean start.
  Finds the running Kafka container by name filter, then loops over the 8 pipeline
  topics deleting each via kafka-topics.sh --delete, waits 1 second, then recreates
  it with 3 partitions and replication-factor 1. Called by restart_aegentra.sh
  when user answers "y" to the purge prompt.

NOTE: Only covers the 8 core pipeline topics (not filtered-out-alerts or
      held-for-review). The --delete + sleep 1 + --create pattern ensures
      Kafka finishes cleanup before recreation.

#!/bin/bash

# Purge all AEGENTRA topics for fresh start

KAFKA_CONTAINER=$(docker ps --filter "name=kafka" --format "{{.Names}}" | head -1)

if [ -z "$KAFKA_CONTAINER" ]; then
    echo "❌ Kafka container not found!"
    exit 1
fi

echo "🧹 Purging all AEGENTRA topics..."

TOPICS=(
    "web-detection-alerts"
    "unified-alerts"
    "correlated-incidents"
    "llm-analysis"
    "response-plans"
    "policy-decisions"
    "executed-actions"
    "audit-logs"
)

for topic in "${TOPICS[@]}"; do
    echo "  Deleting and recreating: $topic"

    # Delete topic
    docker exec $KAFKA_CONTAINER kafka-topics.sh \
        --bootstrap-server localhost:9092 \
        --delete --topic $topic 2>/dev/null

    sleep 1

    # Recreate topic
    docker exec $KAFKA_CONTAINER kafka-topics.sh \
        --bootstrap-server localhost:9092 \
        --create --topic $topic \
        --partitions 3 \
        --replication-factor 1 2>/dev/null

    echo "  ✅ $topic purged"
done

echo ""
echo "🎉 All topics purged! Ready for fresh DVWA scan!"
echo ""
echo "Next steps:"
echo "  1. Scan DVWA: python scanners/zap_scanner.py"
echo "  2. Check topics: http://localhost:8080"

WHAT IT DOES:
  Snapshot tool for the web-detection-alerts Kafka topic. Consumes all messages
  from the earliest offset with a 5-second timeout, saves the full raw JSON list
  to /tmp/scan1_alerts.json, and writes a frequency-sorted summary (count + alert
  name per line) to /tmp/scan1_summary.txt. Used to capture a before-patches
  baseline for comparing vulnerability counts before/after AEGENTRA patches
  are applied.

KEY FLOW:
  1. KafkaConsumer with auto_offset_reset='earliest', consumer_timeout_ms=5000
  2. Drain all available messages into alerts list
  3. Write full list as pretty-printed JSON → /tmp/scan1_alerts.json
  4. Build summary dict: alert_name → count
  5. Write sorted-by-count summary → /tmp/scan1_summary.txt
  6. Print same summary to stdout

#!/usr/bin/env python3
import json
from kafka import KafkaConsumer

print("Reading alerts from Kafka topic 'web-detection-alerts'...")

consumer = KafkaConsumer(
    'web-detection-alerts',
    bootstrap_servers='localhost:9092',
    auto_offset_reset='earliest',
    consumer_timeout_ms=5000,
    value_deserializer=lambda m: json.loads(m.decode('utf-8'))
)

alerts = []
for msg in consumer:
    alerts.append(msg.value)

consumer.close()

# Save detailed JSON
with open('/tmp/scan1_alerts.json', 'w') as f:
    json.dump(alerts, f, indent=2)

# Save summary by type
summary = {}
for alert in alerts:
    key = alert.get('alert', 'Unknown')
    summary[key] = summary.get(key, 0) + 1

with open('/tmp/scan1_summary.txt', 'w') as f:
    f.write(f"SCAN 1 (BEFORE PATCHES): {len(alerts)} total vulnerabilities\n")
    f.write("=" * 80 + "\n")
    for vuln, count in sorted(summary.items(), key=lambda x: x[1], reverse=True):
        f.write(f"{count:3d}x  {vuln}\n")

print(f"\n✅ Saved {len(alerts)} alerts to /tmp/scan1_alerts.json")
print(f"✅ Summary saved to /tmp/scan1_summary.txt")
print("\nScan 1 Summary:")
print("=" * 80)
for vuln, count in sorted(summary.items(), key=lambda x: x[1], reverse=True):
    print(f"{count:3d}x  {vuln}")
print("=" * 80)

WHAT IT DOES:
  Core infrastructure stack for AEGENTRA. Defines 5 services on a shared
  bridge network (aegentra-net):
    - Zookeeper (port 2181) — Kafka coordination
    - Kafka (port 9092) — message broker; internal listener on kafka:29092,
      external on localhost:9092; auto-creates topics enabled
    - PostgreSQL 15 (port 5432) — persistent audit + incident DB;
      mounts init-db.sql for automatic schema creation on first start
    - Redis 7 (port 6379) — caching with AOF persistence + password auth
    - Kafka UI (port 8080) — web dashboard for Kafka topic inspection

  All services use named volumes for data persistence and restart: unless-stopped
  for auto-recovery.

-- Aegentra Database Schema
-- Automatically executed when PostgreSQL container starts

-- Incidents table
CREATE TABLE IF NOT EXISTS incidents (
    id SERIAL PRIMARY KEY,
    incident_id VARCHAR(100) UNIQUE NOT NULL,
    severity VARCHAR(20),
    status VARCHAR(50),
    source_ips TEXT[],
    destination_ips TEXT[],
    attack_type VARCHAR(100),
    mitre_tactics TEXT[],
    mitre_techniques TEXT[],
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    metadata JSONB
);

-- Alerts table
CREATE TABLE IF NOT EXISTS alerts (
    id SERIAL PRIMARY KEY,
    alert_id VARCHAR(100) UNIQUE NOT NULL,
    incident_id VARCHAR(100) REFERENCES incidents(incident_id),
    alert_type VARCHAR(50),
    source_ip VARCHAR(45),
    destination_ip VARCHAR(45),
    source_port INTEGER,
    destination_port INTEGER,
    protocol VARCHAR(20),
    signature VARCHAR(500),
    severity INTEGER,
    raw_data JSONB,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- Audit logs table (immutable)
CREATE TABLE IF NOT EXISTS audit_logs (
    id SERIAL PRIMARY KEY,
    timestamp TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    agent_name VARCHAR(100) NOT NULL,
    incident_id VARCHAR(100),
    action_type VARCHAR(50) NOT NULL,
    evidence JSONB,
    policy_rule VARCHAR(200),
    reasoning_trace TEXT,
    confidence_score FLOAT,
    action_outcome VARCHAR(50),
    metadata JSONB
);

-- Policy rules table
CREATE TABLE IF NOT EXISTS policy_rules (
    id SERIAL PRIMARY KEY,
    rule_name VARCHAR(200) UNIQUE NOT NULL,
    rule_type VARCHAR(50),
    conditions JSONB,
    actions TEXT[],
    enabled BOOLEAN DEFAULT true,
    priority INTEGER DEFAULT 100,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- Create indexes for performance
CREATE INDEX idx_incidents_created_at ON incidents(created_at);
CREATE INDEX idx_incidents_severity ON incidents(severity);
CREATE INDEX idx_alerts_source_ip ON alerts(source_ip);
CREATE INDEX idx_alerts_incident_id ON alerts(incident_id);
CREATE INDEX idx_audit_logs_timestamp ON audit_logs(timestamp);
CREATE INDEX idx_audit_logs_incident_id ON audit_logs(incident_id);
