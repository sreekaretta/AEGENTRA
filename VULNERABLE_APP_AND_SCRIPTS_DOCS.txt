@app.route('/')

@app.route('/login', methods=['GET', 'POST'])
def login():
    """Login page - vulnerable to SQL injection"""
    if request.method == 'POST':
        username = request.form['username']
        password = request.form['password']

        # VULNERABILITY: SQL Injection
        query = f"SELECT * FROM users WHERE username = '{username}' AND password = '{password}'"
        conn = get_db()
        user = conn.execute(query).fetchone()
        conn.close()

        if user:
            session['user_id'] = user['id']
            session['username'] = user['username']
            session['role'] = user['role']
            flash('Login successful!', 'success')
            return redirect(url_for('dashboard'))
        else:
            flash('Invalid credentials', 'error')

    return render_template('login.html')

@app.route('/register', methods=['GET', 'POST'])
def register():
    """Registration page - vulnerable to weak password storage"""
    if request.method == 'POST':
        username = request.form['username']
        password = request.form['password']
        email = request.form['email']

        # VULNERABILITY: Weak password hashing (MD5, no salt)
        hashed_password = hashlib.md5(password.encode()).hexdigest()

        conn = get_db()
        try:
            conn.execute("INSERT INTO users (username, password, email, role) VALUES (?, ?, ?, ?)",
                        (username, hashed_password, email, 'user'))
            conn.commit()
            flash('Registration successful!', 'success')
            return redirect(url_for('login'))
        except sqlite3.IntegrityError:
            flash('Username already exists', 'error')
        finally:
            conn.close()

    return render_template('register.html')

@app.route('/search')
def search():
    """Search page - vulnerable to SQL injection and XSS"""
    query = request.args.get('q', '')

    # VULNERABILITY: SQL Injection in search
    conn = get_db()
    search_query = f"SELECT * FROM products WHERE name LIKE '%{query}%' OR description LIKE '%{query}%'"
    products = conn.execute(search_query).fetchall()

    ip_address = request.remote_addr
    user_agent = request.headers.get('User-Agent', '')
    conn.execute("INSERT INTO search_logs (query, ip_address, user_agent) VALUES (?, ?, ?)",
                (query, ip_address, user_agent))
    conn.commit()
    conn.close()

    return render_template('search.html', query=query, products=products)

FILE 20: scripts/setup_kafka_topics.py

EXPLANATION:
Creates all 10 Kafka topics needed for the full AEGENTRA pipeline. Groups topics
logically: 3 detection-specific (network, web, malware), 1 unified, 6 pipeline
(correlated-incidents through audit-logs). Uses KafkaAdminClient with

from kafka.admin import KafkaAdminClient, NewTopic

def create_topics():
    admin_client = KafkaAdminClient(
        bootstrap_servers='localhost:9092',
        client_id='aegentra-admin'
    )

    topics = [
        # Detection-specific topics (separate by source)
        NewTopic(name='network-detection-alerts', num_partitions=3, replication_factor=1),
        NewTopic(name='web-detection-alerts', num_partitions=3, replication_factor=1),
        NewTopic(name='malware-detection-alerts', num_partitions=3, replication_factor=1),

        # Unified alerts (all sources combined after filtering)
        NewTopic(name='unified-alerts', num_partitions=3, replication_factor=1),

        # Processing pipeline topics
        NewTopic(name='correlated-incidents', num_partitions=2, replication_factor=1),
        NewTopic(name='llm-analysis', num_partitions=2, replication_factor=1),
        NewTopic(name='response-plans', num_partitions=2, replication_factor=1),
        NewTopic(name='policy-decisions', num_partitions=2, replication_factor=1),
        NewTopic(name='executed-actions', num_partitions=1, replication_factor=1),
        NewTopic(name='audit-logs', num_partitions=1, replication_factor=1),
    ]

    try:
        admin_client.create_topics(new_topics=topics, validate_only=False)
        print("✅ Topics created successfully!")
        for topic in topics:
            print(f"  - {topic.name}")
    except Exception as e:
        print(f"Note: {e}")
        print("(Topics may already exist, which is fine)")
    finally:
        admin_client.close()

if __name__ == "__main__":
    create_topics()

TOPIC = 'detection-alerts'

FILE 23: create_summary_table.py
================================================================================
EXPLANATION:
Post-run reporting tool. Reads all 8 pipeline Kafka topics from earliest offset,
cross-references executed-actions with llm-analysis, response-plans, and
policy-decisions by incident_id/plan_id, then builds a summary DataFrame.
Exports to both Excel (/tmp/aegentra_summary.xlsx) and CSV (/tmp/aegentra_summary.csv)
with columns: Incident ID, Execution ID, Threat Type, Severity, Confidence,
MITRE Technique, Actions Total/Successful/Failed, Overall Status, Tech Stack, etc.
--------------------------------------------------------------------------------
#!/usr/bin/env python3
"""Summary Table Generator - Creates Excel/CSV summary of entire pipeline"""

from kafka import KafkaConsumer
import json
import pandas as pd
from datetime import datetime
import os

print("=" * 60)
print("AEGENTRA - Summary Table Generator")
print("=" * 60)

bootstrap_servers = ['localhost:9092']

topics = {
    'web-detection-alerts': [],
    'unified-alerts': [],
    'correlated-incidents': [],
    'llm-analysis': [],
    'response-plans': [],
    'policy-decisions': [],
    'executed-actions': [],
    'audit-logs': []
}

print("\nReading Kafka topics...")

for topic_name in topics.keys():
    try:
        consumer = KafkaConsumer(
            topic_name,
            bootstrap_servers=bootstrap_servers,
            value_deserializer=lambda m: json.loads(m.decode('utf-8')),
            auto_offset_reset='earliest',
            enable_auto_commit=False,
            consumer_timeout_ms=5000
        )
        messages = [message.value for message in consumer]
        topics[topic_name] = messages
        consumer.close()
        print(f"  ✅ {topic_name}: {len(messages)} messages")
    except Exception as e:
        print(f"  ❌ {topic_name}: {e}")

summary_rows = []

for execution in topics['executed-actions']:
    incident_id = execution.get('incident_id')

    analysis = next((a for a in topics['llm-analysis'] if a.get('incident_id') == incident_id), None)
    plan = next((p for p in topics['response-plans'] if p.get('incident_id') == incident_id), None)
    decision = next((d for d in topics['policy-decisions'] if d.get('plan_id') == execution.get('plan_id')), None)

    row = {
        'Incident ID': incident_id,
        'Execution ID': execution.get('execution_id'),
        'Executed At': execution.get('executed_at'),
        'Threat Type': analysis.get('threat_type', 'Unknown') if analysis else 'Unknown',
        'Severity': analysis.get('severity', 0) if analysis else 0,
        'Confidence': analysis.get('confidence', 0) if analysis else 0,
        'MITRE Technique': analysis.get('mitre_technique', '') if analysis else '',
        'MITRE Tactic': analysis.get('mitre_tactic', '') if analysis else '',
        'Actions Total': execution.get('actions_total', 0),
        'Actions Successful': execution.get('actions_successful', 0),
        'Actions Failed': execution.get('actions_failed', 0),
        'Overall Status': execution.get('overall_status'),
        'Decision': decision.get('decision', 'Unknown') if decision else 'Unknown',
        'Tech Stack': f"{plan.get('tech_stack', {}).get('web_server', 'unknown')}/{plan.get('tech_stack', {}).get('language', 'unknown')}" if plan else 'unknown',
        'Automation Approved': execution.get('automation_approved', False),
        'Real Execution': execution.get('real_execution', False)
    }
    summary_rows.append(row)

df = pd.DataFrame(summary_rows)
df = df.sort_values('Executed At', ascending=False)

output_file = '/tmp/aegentra_summary.xlsx'
df.to_excel(output_file, index=False, sheet_name='Execution Summary')

csv_file = '/tmp/aegentra_summary.csv'
df.to_csv(csv_file, index=False)

print(f"\n✅ Excel: {output_file}")
print(f"✅ CSV:   {csv_file}")
print(f"\nTotal Executions: {len(df)}")
print(f"Total Actions:    {df['Actions Total'].sum()}")
